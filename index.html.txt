<script>
  let records = JSON.parse(localStorage.getItem('proxy_app_records_v2') || '[]');

  async function fetchRates() {
    try {
      const res = await fetch('https://open.er-api.com/v6/latest/TWD');
      const data = await res.json();
      if (data && data.rates) {
        document.getElementById('rateJPY').value = (1 / data.rates.JPY).toFixed(4);
        document.getElementById('rateKRW').value = (1 / data.rates.KRW).toFixed(4);
        updateCalc();
        alert('✅ 匯率更新成功！');
      }
    } catch (e) {
      alert('⚠️ 匯率抓取失敗，請手動輸入。');
    }
  }

  function updateCalc() {
    const currency = document.getElementById('currency').value;
    const foreignCost = parseFloat(document.getElementById('foreignCost').value) || 0;
    const quoteRate = parseFloat(document.getElementById('quoteRate').value) || 0;
    const cardFeePercent = parseFloat(document.getElementById('cardFee').value) || 0;
    const bankRate = parseFloat(document.getElementById(currency === 'JPY' ? 'rateJPY' : 'rateKRW').value) || 0;
    const isTaxChecked = document.getElementById('calcTax').checked;

    const customerPrice = Math.round(foreignCost * quoteRate);
    const actualCost = Math.round(foreignCost * bankRate * (1 + cardFeePercent / 100));
    
    // 計算 5% 營業稅（內含稅公式）
    const tax = isTaxChecked ? Math.round(customerPrice - (customerPrice / 1.05)) : 0;
    const netProfit = customerPrice - actualCost - tax;

    document.getElementById('calcCustomerPrice').innerText = `NT$ ${customerPrice}`;
    document.getElementById('calcActualCost').innerText = `NT$ ${actualCost}`;
    document.getElementById('calcTaxAmount').innerText = `NT$ ${tax}`;
    document.getElementById('calcNetProfit').innerText = `${netProfit >= 0 ? '+' : ''} NT$ ${netProfit}`;

    return { customerPrice, actualCost, tax, netProfit };
  }

  function addPurchaseRecord() {
    const name = document.getElementById('itemName').value || '代購商品';
    const calc = updateCalc();
    const isTaxChecked = document.getElementById('calcTax').checked;
    
    records.push({
      id: Date.now(),
      type: '代購',
      detail: `${name}${isTaxChecked ? '(含5%稅)' : ''}`,
      income: calc.customerPrice,
      deduction: calc.actualCost + calc.tax,
      profit: calc.netProfit
    });

    saveAndRender();
  }

  function addSalesRecord() {
    const platform = document.getElementById('platform').value;
    const amount = parseFloat(document.getElementById('salesAmount').value) || 0;
    const fee = parseFloat(document.getElementById('platformFee').value) || 0;

    if (amount <= 0) return alert('請輸入銷售金額');

    let netSales = amount - fee;

    records.push({
      id: Date.now(),
      type: '電商',
      detail: platform,
      income: amount,
      deduction: fee,
      profit: netSales
    });

    saveAndRender();
    document.getElementById('salesAmount').value = '';
  }

  function deleteRecord(id) {
    records = records.filter(r => r.id !== id);
    saveAndRender();
  }

  function saveAndRender() {
    localStorage.setItem('proxy_app_records_v2', JSON.stringify(records));
    
    const tbody = document.getElementById('recordTable');
    tbody.innerHTML = '';

    let totalProxyProfit = 0;
    let totalSalesIncome = 0;

    records.forEach(r => {
      if (r.type === '代購') totalProxyProfit += r.profit;
      if (r.type === '電商') totalSalesIncome += r.income;

      const tr = document.createElement('tr');
      tr.innerHTML = `
        <td><b>${r.type}</b></td>
        <td>${r.detail}</td>
        <td>$${r.income}</td>
        <td>$${r.deduction}</td>
        <td style="color: ${r.profit >= 0 ? '#059669' : '#ef4444'}; font-weight: bold;">$${r.profit}</td>
        <td><button class="delete-btn" onclick="deleteRecord(${r.id})">刪除</button></td>
      `;
      tbody.appendChild(tr);
    });

    document.getElementById('summaryText').innerText = 
      `代購淨利：NT$ ${totalProxyProfit.toLocaleString()} | 電商銷售額：NT$ ${totalSalesIncome.toLocaleString()}`;
  }

  updateCalc();
  saveAndRender();
</script>
